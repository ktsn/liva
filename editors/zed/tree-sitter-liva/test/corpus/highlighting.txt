motion card(active: boolean) {
  layer base { body @tween(duration: 200) { opacity: 1 } }
  layer hover when active { body { scale: 1.1 @spring(stiffness: 300) } }
  layer machine {
    state idle initial {}
    state busy {}
    transition idle -> busy when active
    transition busy -> idle when !active { body { scale @tween(duration: 150) { 50%: 1.2 @easeOut } } }
  }
  enter { body @tween(duration: 200) { opacity: 0 } }
}

motion sample(enabled: boolean) {
  layer base @tween(duration: 200) { body { opacity: 1 } }
  layer machine @spring(duration: 400) {
    state idle initial @tween() {}
    state active @tween(easing: steps(count: 4)) {}
    transition idle -> active
      @spring(duration: 300, bounce: 0.2)
      when enabled
  }
  enter @tween(duration: 100) { body { opacity: 0 } }
}
