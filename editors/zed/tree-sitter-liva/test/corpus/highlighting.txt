motion swipe {
  let pointer = drag(target: self)
  let movement = link(input: pointer.offset.x, component: "x")
  layer position {
    state closed initial {}
    state open {}
    on pointer.release {
      transition closed -> open when movement.value.x > 48
      transition open -> closed
    }
  }
  layer dragging when pointer.active {
    translate := movement
  }
}
